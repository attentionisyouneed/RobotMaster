#include <iostream>
 
void reverseArray(int* arr, int n) {
    if (arr == nullptr || n <= 0) {
        return; 
    }
    int* left = arr;
    int* right = arr + n - 1;
    while (left < right) {
        int temp = *left;
        *left = *right;
        *right = temp;
        left++;  
        right--; 
    }
}
 
int main() {
    int n;
    if (!(std::cin >> n)) {
        return 0;
    }
    if (n < 0 || n > 100) {
        return 0;
    }
    int arr[100];
    for (int i = 0; i < n; i++) {
        std::cin >> arr[i];
    }
    reverseArray(arr, n);
    for (int i = 0; i < n; i++) {
        if (i > 0) std::cout << " ";
        std::cout << arr[i];
    }
    std::cout << std::endl;
    return 0;
}

#include <iostream>
#include <chrono>
 
long long climbStairsIter(int n) {
    if (n < 0) return 0;
    if (n == 0 || n == 1) return 1; 
    long long a = 1; 
    long long b = 1; 
    long long c = 0; 
    for (int i = 2; i <= n; i++) {
        c = a + b;   
        a = b;       
        b = c;
    }
    return c;
}
 
long long climbStairsRecur(int n) {
    if (n < 0) return 0;
    if (n == 0 || n == 1) return 1; 
    return climbStairsRecur(n - 1) + climbStairsRecur(n - 2);
}
 
int main() {
    int n;
    if (!(std::cin >> n)) {
        return 0;
    }
    if (n < 0 || n > 40) {
        std::cout << "阶数不合法，请输入 0 到 40。" << std::endl;
        return 0;
    }
    auto startIter = std::chrono::high_resolution_clock::now();
    long long resIter = climbStairsIter(n);
    auto endIter = std::chrono::high_resolution_clock::now();
    std::chrono::duration<double, std::milli> elapsedIter = endIter - startIter;
    auto startRecur = std::chrono::high_resolution_clock::now();
    long long resRecur = climbStairsRecur(n);
    auto endRecur = std::chrono::high_resolution_clock::now();
    std::chrono::duration<double, std::milli> elapsedRecur = endRecur - startRecur;
    std::cout << "\n========== 运行结果 ==========" << std::endl;
    std::cout << "迭代实现结果: " << resIter << " (耗时: " << elapsedIter.count() << " ms)" << std::endl;
    std::cout << "递归实现结果: " << resRecur << " (耗时: " << elapsedRecur.count() << " ms)" << std::endl;
    if (resIter == resRecur) {
        std::cout << "对比结论: 两种方法计算结果完全一致！" << std::endl;
    } else {
        std::cout << "对比结论: 结果不一致（程序可能存在 bug）！" << std::endl;
    }
    std::cout << "==============================" << std::endl;
    if (n >= 35) {
        std::cout << "\n💡 提示: 当 n >= 35 时，递归由于存在大量的重复计算，耗时会明显上升。" << std::endl;
        std::cout << "而迭代方法时间复杂度为 O(n)，无论 n 多大都是瞬间完成。" << std::endl;
    }
    return 0;
}

void preorderTraversal(const int* tree, int n, int i) {
    if (i >= n || tree[i] == -1) {
        return;
    }
    std::cout << tree[i] << " ";                 
    preorderTraversal(tree, n, 2 * i + 1);       
    preorderTraversal(tree, n, 2 * i + 2);       
}
 
void inorderTraversal(const int* tree, int n, int i) {
    if (i >= n || tree[i] == -1) {
        return;
    }
    inorderTraversal(tree, n, 2 * i + 1);       
    std::cout << tree[i] << " ";                 
    inorderTraversal(tree, n, 2 * i + 2);       
}
 
void postorderTraversal(const int* tree, int n, int i) {
    if (i >= n || tree[i] == -1) {
        return;
    }
    postorderTraversal(tree, n, 2 * i + 1);      
    postorderTraversal(tree, n, 2 * i + 2);      
    std::cout << tree[i] << " ";                 
}
 
void buildTestTree(int* tree, int n) {
    for (int i = 0; i < n; i++) {
        tree[i] = -1;
    }
    if (n > 0) tree[0] = 1; 
    if (n > 1) tree[1] = 2; 
    if (n > 2) tree[2] = 3; 
    if (n > 3) tree[3] = 4; 
    if (n > 4) tree[4] = 5; 
}
 
int main() {
    const int TREE_SIZE = 7;

    buildTestTree(tree, TREE_SIZE);
    std::cout << "========== 二叉树遍历结果 ==========" << std::endl;

    std::cout << "\n(期望结果: 4 2 5 1 3)\n" << std::endl;
    std::cout << "3. 后序遍历 (左 -> 右 -> 根): ";

#include <iostream>
 
const int MAX_STUDENTS = 100;
 
void initScores(int* scores, int n) {
    for (int i = 0; i < n; i++) {
        scores[i] = -1;
    }
}
 
void inputScores(int* scores, int n) {
    std::cout << "\n--- 开始录入成绩 ---" << std::endl;
    for (int i = 0; i < n; i++) {
        int score;
        while (true) {
            std::cout << "请输入学生 " << i << " 的成绩 (0-100): ";
            std::cin >> score;
            if (score >= 0 && score <= 100) {
                scores[i] = score;
                break;
            } else {
                std::cout << "⚠️ 输入成绩不合法 (必须在 0 到 100 之间)，请重新录入！" << std::endl;
            }
        }
    }
    std::cout << "--- 成绩录入完成 ---\n" << std::endl;
}
 
double calcAverage(const int* scores, int n) {
    if (n <= 0) return 0.0;
    double sum = 0.0;
    int validCount = 0;
    for (int i = 0; i < n; i++) {
        if (scores[i] != -1) { 
            sum += scores[i];
            validCount++;
        }
    }
    if (validCount == 0) return 0.0;
    return sum / validCount;
}
 
int safeGet(const int* scores, int n, int index) {
    if (index < 0 || index >= n) {
        std::cout << "⚠️ 下标越界 (当前有效范围: 0 至 " << n - 1 << ")" << std::endl;
        return -1;
    }
    if (scores[index] == -1) {
        std::cout << "⚠️ 下标 " << index << " 合法，但该位置的学生成绩未录入！" << std::endl;
        return -1;
    }
    return scores[index];
}
 
void countGrades(const int* scores, int n, int* counts) {
    for (int i = 0; i < 5; i++) {
        counts[i] = 0;
    }
    for (int i = 0; i < n; i++) {
        int s = scores[i];
        if (s == -1) continue; 
        if (s >= 90 && s <= 100) {
            counts[0]++;
        } else if (s >= 80 && s < 90) {
            counts[1]++;
        } else if (s >= 70 && s < 80) {
            counts[2]++;
        } else if (s >= 60 && s < 70) {
            counts[3]++;
        } else if (s >= 0 && s < 60) {
            counts[4]++;
        }
    }
}
 
void findMinMax(const int* scores, int n, int* max, int* min) {
    if (n <= 0) {
        *max = -1;
        *min = -1;
        return;
    }
    int currentMax = -1;
    int currentMin = 101;
    int validCount = 0;
    for (int i = 0; i < n; i++) {
        if (scores[i] != -1) {
            if (scores[i] > currentMax) {
                currentMax = scores[i];
            }
            if (scores[i] < currentMin) {
                currentMin = scores[i];
            }
            validCount++;
        }
    }
    if (validCount == 0) {
        *max = -1;
        *min = -1;
    } else {
        *max = currentMax;
        *min = currentMin;
    }
}
 
void printResults(const int* scores, int n, double avg, const int* counts, int max, int min) {
    std::cout << "\n========== 统计结果 ==========" << std::endl;
    std::cout << "学生总人数: " << n << " 人" << std::endl;
    std::cout << "成绩列表: ";
    for (int i = 0; i < n; i++) {
        if (scores[i] == -1) {
            std::cout << "未录入 ";
        } else {
            std::cout << scores[i] << " ";
        }
    }
    std::cout << std::endl;
    std::cout << "班级平均分: " << avg << " 分" << std::endl;
    std::cout << "最高分: " << (max == -1 ? 0 : max) << " 分" << std::endl;
    std::cout << "最低分: " << (min == -1 ? 100 : min) << " 分" << std::endl;
    std::cout << "分数段人数统计:" << std::endl;
    std::cout << "  [90 - 100 分 (优秀)]: " << counts[0] << " 人" << std::endl;
    std::cout << "  [80 - 89 分  (良好)]: " << counts[1] << " 人" << std::endl;
    std::cout << "  [70 - 79 分  (中等)]: " << counts[2] << " 人" << std::endl;
    std::cout << "  [60 - 69 分  (及格)]: " << counts[3] << " 人" << std::endl;
    std::cout << "  [0 - 59 分 (不及格)]: " << counts[4] << " 人" << std::endl;
    std::cout << "==============================" << std::endl;
}
 
int main() {
    int scores[MAX_STUDENTS];
    int n;
    int counts[5] = {0};
    int max = -1, min = -1;
    double avg = 0.0;
    int testIndex;
    initScores(scores, MAX_STUDENTS);
    std::cout << "请输入学生人数 (1-" << MAX_STUDENTS << "): ";
    std::cin >> n;
    if (n <= 0 || n > MAX_STUDENTS) {
        std::cout << "输入人数不合法！" << std::endl;
        return 0;
    }
    inputScores(scores, n);
    avg = calcAverage(scores, n);
    findMinMax(scores, n, &max, &min);
    countGrades(scores, n, counts);
    std::cout << "请输入要查询的下标：";
    std::cin >> testIndex;
    int queried = safeGet(scores, n, testIndex);
    if (queried == -1) {
        std::cout << "查询失败" << std::endl;
    } else {
        std::cout << "下标 " << testIndex << " 的成绩是：" << queried << std::endl;
    }
    printResults(scores, n, avg, counts, max, min);
    return 0;
}
